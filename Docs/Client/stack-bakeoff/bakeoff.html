ExcelJS — sel bisa balik sebagai objek

function plain(v) {
  if (typeof v === 'object') {
    if ('error' in v) return v.error;      // '#DIV/0!'
    if ('result' in v) return v.result;    // sel formula
    if ('richText' in v)
      return v.richText.map(t => t.text).join('');
    if ('text' in v) return v.text;        // hyperlink
  }
  return v;
}
